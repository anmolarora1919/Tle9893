Description of the Full Code : 
This code is running 6steps algorithm to run a BLDC motor with the help of hall sensors. 
The speed request is taken from some ADC input, must there be a potentiometer on the eval board at pin P2.4.
This code is running on Interrupts.
1 - Main_HandleSysTick : This interrupt might be coming every 1msec. This timer increments the duty cycle of the PWM with a PID algorithm, which helps to attain the required speed. Input from potentiometer is read here only.
2 - EmoCcu_HandleHallEvent : This interrupt comes at every correct/expected hall sensor combination is found. Here we turn on or off the MosFETs which are required for the next step as per the 6 step table. present speed calculation is done in this interrupt only.


- It is to note that, the Sine wave is generated on the PWM pins and not on the Mosfet Driver Pins. : So BDRV register are not used in this code and are commented.

void BDRV_Init(void)
{
  /* Set charge pump clock */
  BDRV->CP_CLK_CTRL.reg = (uint32) BDRV_CP_CLK_CTRL;
  /* Set charge pump control, clear status bits */
  BDRV->CP_CTRL_STS.reg = (uint32) BDRV_CP_CTRL_STS;
  /* disable all interrupts                */
  CMSIS_Irq_Dis();
  /* Set trimming register */
  SCU_OpenPASSWD();
  BDRV->TRIM_DRVx.reg = (uint32) BDRV_TRIM_DRVx;
  SCU_ClosePASSWD();
  /* enable all interrupts                 */
  CMSIS_Irq_En();
  /* Set bridge driver control 1/2/3 */
  BDRV->CTRL1.reg = (uint32) BDRV_CTRL1;
  BDRV->CTRL2.reg = (uint32) BDRV_CTRL2;
  BDRV->CTRL3.reg = (uint32) BDRV_CTRL3;
#if (CONFIGWIZARD == 1)
  BDRV->OFF_SEQ_CTRL.reg = (uint32) BDRV_OFF_SEQ;
  BDRV->ON_SEQ_CTRL.reg = (uint32) BDRV_ON_SEQ;
#else /* (CONFIGWIZARD == 2) */
  BDRV->OFF_SEQ_CTRL.reg = (uint32) BDRV_OFF_SEQ_CTRL;
  BDRV->ON_SEQ_CTRL.reg = (uint32) BDRV_ON_SEQ_CTRL;
#endif
#if (UC_SERIES == TLE987)
  MF->BEMFC_CTRL_STS.reg = (uint32) MF_BEMFC_CTRL_STS;
  MF->TRIM_BEMFx.reg = (uint32) MF_TRIM_BEMFx;
#endif
} /* End of BDRV_Init */

uint32 Emo_Init(void)
{
  if (Emo_Status.MotorState != EMO_MOTOR_STATE_UNINIT)
  {
    /* Error detected: return with error */
    return EMO_ERROR_MOTOR_INIT;
  }

  /* Switch on VDDEXT */
  PMU->VDDEXT_CTRL.reg = 0x01u;
  /* Initialize Hall parameters */
  EmoCcu_InitHallPar();
  /* Initialize parameters */
  Emo_lInitPar();
  /* Initialize motor state */
  Emo_Status.MotorState = EMO_MOTOR_STATE_STOP;
  /* Return without error */
  return EMO_ERROR_NONE;
} /* End of Emo_Init */

void Poti_Handler(void)
{
  uint16 mV;

  /* Read the value at Ch4 (Poti) in mV - values between 0 and 5000 are possible */
  if (ADC1_GetChResult_mV(&mV, ADC1_CH4) == true)
  {
    Emo_SetRefSpeed(mV / 2);

    if (mV > 100)
    {
      Main_lStartMotor();			// Definition below : Start the motor in 1 direction.
      PORT_P01_Output_High_Set();
    }
    else
    {
      Main_lStopMotor();			// Definition below : Stop the motor.
      PORT_P01_Output_Low_Set();
    }
  }
}

uint32 Emo_StartMotor(void)
{
  if (Emo_Status.MotorState != EMO_MOTOR_STATE_STOP)
  {
    /* Error detected: return with error */
    return EMO_ERROR_MOTOR_NOT_STOPPED;
  }

  /* Start PWM */
  Ccu6_Start();								//Definition below : This is from Infineon drivers, which helps to start the PWM 									//using CCU6.	
  /* Enable bridge (after starting PWM according to UM) */
  BDRV_Set_Bridge(Ch_PWM, Ch_PWM, Ch_PWM, Ch_PWM, Ch_PWM, Ch_PWM);	//This is an important driver level function which helps us to start the PWM signal 									//on the Mosfet driver. This is enabled after enabling the driver.
  /* Initialize variables */
  Emo_lInitVar();
  /* Initialize Hall variables */
  EmoCcu_InitHallVar();
  /* Set start state */
  Emo_Status.MotorState = EMO_MOTOR_STATE_START;
  /* Return without error */
  return EMO_ERROR_NONE;
} /* End of Emo_StartMotor */

This is from Infineon drivers, which helps to start the PWM using CCU6.
--------------------------------------------------------------------------------
void Ccu6_Start(void)
{
  uint32 HallOutPtn;
  uint16 InitDutyCycle;
  uint32 CurrentHallPtn;
  uint32 ExpHallPtn;
  /* Set common duty cycle */
  InitDutyCycle = (uint16)((((uint32)BCHALL_INIT_DUTY) * EMO_PWM_PERIOD_TICKS) / 100);
  CCU6_LoadShadowRegister_CC60(InitDutyCycle);
  CCU6_LoadShadowRegister_CC61(InitDutyCycle);
  CCU6_LoadShadowRegister_CC62(InitDutyCycle);
  /* Timer13 triggers the HALL pattern evaluation, wait 50µs until TIMER13 triggers HALL evaluation */
  Delay_us(100u);
  /* Get current Hall pattern */
  CurrentHallPtn = CCU6_ReadHallReg();
  /* Set current/next T12MODEN and MCMEN in CCU60_MODCTR */
  CCU6_ConfigureGlobalModulation((uint16)EmoCcu_Cfg.T12Moden[CurrentHallPtn + EmoCcu_HallStatus.DirIdx]);
  /* Get current Hall and output patterns from table */
  HallOutPtn = EmoCcu_Cfg.HallOutPtns[CurrentHallPtn + EmoCcu_HallStatus.DirIdx];
  /* Set current Hall and output patterns immediately in MCMOUT */
  CCU6_WriteMultichannelPatterns((uint16)(HallOutPtn | CCU6_MASK_MCMOUTS_SHADOW_OUT | CCU6_MASK_MCMOUTS_SHADOW_HALL));
  /* Prepare next Hall and output patterns in CCU6_MCMOUTS */
  ExpHallPtn = (HallOutPtn >> 8u) & 0x7u;
  CCU6_WriteMultichannelPatterns(EmoCcu_Cfg.HallOutPtns[ExpHallPtn + EmoCcu_HallStatus.DirIdx]);
  /* Set Multi-Channel Mode Control register for switching on correct Hall pattern */
  CCU6_ConfigureMultichannelModulation((uint16)Ccu6_lSetMCMCTR
                                       (
                                         1, /* SWSEL Switching Selection(3 Bits):
        * 0=No trigger request, 1=Correct Hall pattern, 2=T13 period match, 3=T12 one match,
        * 4=T12 channel 1 compare match, 5=T12 period match, 6..7=reserved */
                                         0, /* SWSYN Switching Synchronization(2 Bits):
        * 0=Direct, 1=T13 zero-match, 2=T12 zero-match (while counting up), 3=reserved */
                                         0, /* STE12U Shadow Transfer Enable for T12 Upcounting:
        * 1=Enable transfer for T12 upcounting and MCMEN=1 */
                                         0, /* STE12D Shadow Transfer Enable for T12 Downcounting:
        * 1=Enable transfer for T12 downcounting and MCMEN=1 */
                                         0  /* STE13U Shadow Transfer Enable for T13 Upcounting:
        * 1=Enable transfer for T13 upcounting and MCMEN=1 */
                                       ));
  /* Start timer 3 */
  GPT12E_T3_Start();
  /* Clear status bits, enable interrupt for Hall event only */
  CCU6_ClearIntStatus(CCU6_MASK_INT_CHE);
  CCU6_EnableInt(CCU6_MASK_INT_CHE);
  /* Start T12, enable shadow transfer for T12 and T13 */
  CCU6_SetT12T13ControlBits((uint16)(CCU6_MASK_TCTR4_START_T12 | CCU6_MASK_TCTR4_SHADOW_T12 | CCU6_MASK_TCTR4_SHADOW_T13));
} /* End of Ccu6_Start */

uint32 Emo_StopMotor(void)
{
  if (Emo_Status.MotorState < EMO_MOTOR_STATE_START)
  {
    /* Error detected: return with error */
    return EMO_ERROR_MOTOR_NOT_STARTED;
  }

  /* Disable bridge */
  BDRV_Set_Bridge(Ch_Off, Ch_Off, Ch_Off, Ch_Off, Ch_Off, Ch_Off);		// All Bridge Drivers are turned off.
  /* Stop PWM */
  Ccu6_Stop();									// CCU6 PWM is stopped. This is the sequence to turn off the Mosfet Drivers.
  /* Initialize bridge */
  BDRV_Init();									// Bridge driver is again initiated.
  /* Set stop state */
  Emo_Status.MotorState = EMO_MOTOR_STATE_STOP;					// Motor state is turned off.
  /* Return without error */
  return EMO_ERROR_NONE;
} /* End of Emo_StopMotor */

void EmoCcu_HandleHallEvent(void)
{
  uint32 ExpHallPtn;
  uint32 CurrentHallPtn;
  uint32 HallPtn;
  uint16 DiffTime;
  uint16 Speed;
  uint16 Time;
  uint16 DelayTime;
  /* Set current/next T12MODEN and MCMEN in CCU60_MODCTR */
MCMOUT register value is returned.
  HallPtn = (uint32)(CCU6_ReadMultichannelPatterns() >> 8u);				// Reading register(MCMOUT) values which contains Mosfet states and 											// Hall Sensor States and shifting it with 8 values to right.

  CurrentHallPtn = HallPtn >> 3u;							// It looks they are trying to get the current hall pattern. Present 											// hall Sensor value is at bit 11,12,13. Shifting 8+3 will provide 											// its value.

  CCU6_ConfigureGlobalModulation((uint16)EmoCcu_Cfg.T12Moden[CurrentHallPtn + EmoCcu_HallStatus.DirIdx]);	// Definition below. This function enables 														// which Mosfets will remain connected with 														// CCU6/PWM output and which will not based 														// on the hall pattern using register 															// "MODCTR"
  /* Prepare next Hall patterns and output pattern */
  ExpHallPtn = HallPtn & 0x7u;								// Fetching the Hall Pattern from the variable and removing other 											// values.
  
  CCU6_WriteMultichannelPatterns(EmoCcu_Cfg.HallOutPtns[ExpHallPtn + EmoCcu_HallStatus.DirIdx]);	// Definition below : Below function writes to the 													// shadow register which writes in the "expected 													// Hall sensor values" for the next cycles and PWM 													// enabling shadow register, which Output needs to 													// be connected to the CCU6(PWM) output.

  Time = GPT12E_T3_Value_Get();										// This is the values of another free running timer 													// used to get time value

  if (EmoCcu_HallStatus.StartCtr < 255u)								// Some counter value is checked with maximum value.
  {
    /* Increment start counter */
    EmoCcu_HallStatus.StartCtr++;									// Counter is incremented at every hall interrupt.
  }

  if (EmoCcu_HallStatus.StartCtr == 1u)									// If counter was incremented for the first time.
  {
    /* Start: */											// Do nothing.
    /* Do not calculate difference time and speed */
  }
  else
  {
    /* Calculate difference time */
    DiffTime = Time - EmoCcu_HallStatus.EventTime;							// getting the time difference from last time when 													// hall interrupt came.
    /* Calculate speed */
    Speed = (uint16)(SPEED_FROM_COMM_PERIOD_NUM / DiffTime);						// Calculating the speed from the Time difference.
    EmoCcu_HallStatus.Speed = Speed;									// updating speed in structure variable.

    if (EmoCcu_HallStatus.StartCtr == 2u)								// If counter was 2
    {
      /* Initialize average speed */
      EmoCcu_HallStatus.Speed = Mat_ExeSimpleLp(&EmoCcu_HallStatus.SpeedLong, Speed, 65535u);		// Definition below : We are updating the speed of 													// with low pass filter in the structure variable.
      /* Set run mode */
      Emo_SetMotorState(EMO_MOTOR_STATE_RUN);								// This function just updates the state of Motor 													// State to RUN
    }
    else /* (EmoCcu_HallStatus.StartCtr >= 2u) */							// If the counter value is more than 2.
    {
      /* Calculate average speed */
      EmoCcu_HallStatus.Speed = Mat_ExeSimpleLp(&EmoCcu_HallStatus.SpeedLong, Speed, POS_SPEED_LP_COEF);	// Again the this function provides rounded 														// off average speed.  
    }

    if (Speed >= EmoCcu_HallStatus.DelayMinSpeed)							// If speed is greater than the minimum speed.
    {
      /* Minimum speed reached: */
      /* Set T13 period to Hall delay time, limit to minimum = Hall filter time */			 
      DelayTime = (uint16)(((((uint32)DiffTime) * EmoCcu_HallStatus.DelayAngle) + 30u) / 60u);		// If there is a hall angle delay, it can be 														// compensated here.

      if (DelayTime < T13_HALL_FILTER_TIME_TICKS)							// if Delay is less than Time Period of T13
      {
        DelayTime = T13_HALL_FILTER_TIME_TICKS;								// Delay time becomes equal to Time period of T13,													// Note : T13 timer is not 3 PWM timer, T12 is 3 PWM 													// timer.
      }

      CCU6_LoadPeriodRegister_T13_Tick(DelayTime);			// Defition below : We are setting the period of T13 Timer 												// with Delay time.
							
      CCU6_EnableST_T13();						// Definition below : This function requests shadow transfer 												// of the T13 Timer Time Period.
      EmoCcu_HallStatus.DelayTime = DelayTime;				// Delay time is copied in the Structure delay time.
    }
  }

  EmoCcu_HallStatus.EventTime = Time;					// The time when this interrupt occurred is saved in structure time. 
} /* End of EmoCcu_HandleHallEvent */

This function is a low pass filter which taken input of 16bit, and with the help of a factor it gives a 32bit value.
This function is providing the round off speed.
--------------------------------------------------------------------------------
EmoCcu_HallStatus.Speed = Mat_ExeSimpleLp(&EmoCcu_HallStatus.SpeedLong, Speed, POS_SPEED_LP_COEF);

uint16 Mat_ExeSimpleLp(uint32 *pOutput, uint16 Input, uint16 Fac)
{
  uint32 NegFac;
  uint32 Out;
  Out = *pOutput;
  /* Output = output - output * factor */
  NegFac = 0x10000u - (uint32)Fac;
  Out = (((Out & 0xFFFFu) * NegFac) >> 16u) + ((Out >> 16u) * NegFac);
  /* Output += input * factor */
  Out += (uint32)Input * (uint32)Fac;
  /* Store output */
  *pOutput = Out;
  /* Return upper rounded part */
  return (uint16)((Out + 0x8000u) >> 16u);
}

This function helps to write any register(first argument), with a value "val"(fourth argument). It is also possible to mask specific values using the 3rd argument. It is also possible to shift the value left shift the value using the 2nd argument.
Note : First shifting will be done then masking will happen.
--------------------------------------------------------------------------------
INLINE void Field_Wrt16(volatile uint16 *reg, uint16 pos, uint16 msk, uint16 val)
{
  (*reg = (uint16)((uint32)val << pos) & (uint16)msk);
}

{
  uint16 DutyCycle;							// Create a variable for DutyCycle

  if (Emo_Status.MotorState == EMO_MOTOR_STATE_RUN)			// If the motor state is run.
  {
    /* Perform speed control */
    DutyCycle = (uint16)Mat_ExePi(&Emo_Ctrl.SpeedPi, Emo_Ctrl.UserRefSpeed - (sint16)Emo_GetAbsSpeed());	// Generate duty based on PID.
    /* Set new common duty cycle immediately */
    CCU6_LoadShadowRegister_CC60(DutyCycle);				// Definition below.

    CCU6_LoadShadowRegister_CC62(DutyCycle);
    /* Enable shadow transfer for T12 */
    CCU6_EnableST_T12();						// Definition below : 
    /* Save new duty cycle */
    Emo_Ctrl.DutyCycle = DutyCycle;
  }
} /* End of Emo_CtrlSpeed */

Below function is to calculate the duty cycle using PID algorithm, considering the present speed and the target speed. 
First argument is the structure variable of the type "TMat_Pi", description below. 2nd variable is the error in the speed(diff in requested speed and present speed.) Requested speed is taken from some ADC.
--------------------------------------------------------------------------------
DutyCycle = (uint16)Mat_ExePi(&Emo_Ctrl.SpeedPi, Emo_Ctrl.UserRefSpeed - (sint16)Emo_GetAbsSpeed());	

typedef struct
{
  sint16 UserRefSpeed;  /**< \brief User reference speed [rpm] */
  TMat_Pi SpeedPi;      /**< \brief Speed PI control */
  uint16 DutyCycle;     /**< \brief Duty cycle [PWM timer ticks] */
} TEmo_Ctrl;
   
typedef struct
{
  sint32 IOut;     /**< \brief I output */
  sint16 Kp;       /**< \brief Proportional parameter */
  sint16 Ki;       /**< \brief Integral parameter */
  sint16 IMin;     /**< \brief Minimum for I output */
  sint16 IMax;     /**< \brief Maximum for I output */
  sint16 PiMin;    /**< \brief Minimum for PI output */
  sint16 PiMax;    /**< \brief Maximum for PI output */
} TMat_Pi;


__STATIC_INLINE sint16 Mat_ExePi(TMat_Pi *pPi, sint16 Error)
{
  sint32 IOut;
  sint32 PiOut;
  sint32 Min;
  sint32 Max;
  sint32 Temp;
  /* I output = old output + error * I parameter */
  IOut = pPi->IOut + ((sint32)Error * (sint32)pPi->Ki);		// Iout is found by adding previous value of Output and Present Error multiplied by Ki.
								// So this seems to be the integral element.
  /* Limit I output */
  Min = ((sint32)(pPi->IMin)) << 15u;				// Pi Min value is shifted 15 times, can be also said that got multiplied by 32k...

  if (IOut < Min)						// If Iout is < Min,
  {
    IOut = Min;							// Make Iout equal to Min.
  }
  else
  {
    Max = ((sint32)(pPi->IMax)) << 15u;				// Max value is left shifted by 15.

    if (IOut > Max)						// If I out is greater than max,
    {
      IOut = Max;						// Iout is made equal to max.
    }
  }

  /* Store I output */
  pPi->IOut = IOut;						// Present values of the Pi Iout are saved in the structure variable.

  /* PI output = upper half of (I output + saturate(error * P parameter) * 64) */
  Temp = __SSAT(Error * ((sint32)pPi->Kp), 31u - 6u);		// Error is multiplied by Kp and saturated to 2^25 signed value.
								// Note : Saturation means that, the value go outside the range of signed 2^25 value or it 								// will be clipped. 
  PiOut = (IOut + (Temp << 6u)) >> 15u;				// We add the Proportional value in the present output value and then left shift it 15 									// times.
  /* Limit PI output */
  Min = (sint32)(pPi->PiMin);					//Min value is fetched from the structure. It has come based on GUI. // Value is 8.

  if (PiOut < Min)						// If PiOut is less than Min, we clamp it with Min value.
  {
    PiOut = Min;
  }
  else
  {
    Max = (sint32)(pPi->PiMax);					// Max value is fetched from GUI, which is 75.

    if (PiOut > Max)						// Piout is clamped with max value.
    {
      PiOut = Max;
    }
  }

  return (sint16)PiOut;						// return Piout value.
} /* End of Mat_ExePi */
